export class DevisListRowComponent {
  readonly devisListItemData = input<DevisListItemModel | null>(null);

  private readonly router = inject(Router);

  protected readonly montantRAC = computed(() =>
    MathUtils.convertDecimal(this.devisListItemData()?.montantRAC ?? 0),
  );

  protected readonly montantTotal = computed(() =>
    MathUtils.convertDecimal(this.devisListItemData()?.montantTotal ?? 0),
  );

  protected readonly montantAriaLabel = computed(
    () => `${this.montantRAC()} € sur ${this.montantTotal()} €`,
  );

  protected onClickCard(): void {
    const noDossier = this.devisListItemData()?.noDossier;

    if (!noDossier) {
      return;
    }

    void this.router.navigate(['sante', 'historique', 'details-demande-page', 'devis', noDossier]);
  }
}
